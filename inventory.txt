# ./inventories/pve/hosts.yml
#
# define the host PVE servers that support Proxmox Backup Server virtual machines
all:
  children:
    pve:
      hosts:
        lala100:
          # The host for the primary PBS server
          ansible_host: 10.0.0.100
        lala150:
          # the host for the secondary PBS server
          ansible_host: 10.0.0.150

-------------------------------------------------------------------------------

# ./inventories/pve/group_vars/pve.yml
#
# define the virtual machines that contain the PBS entities.

pve_pbs_vms:
- vmid: 109 # define the primary PBS instance on lala100
  name: pbsfront
  target_node: lala100 # Host PVE machine name for VM
  memory: 2048
  cores: 2
  disks:
  - size: 30G # VM boot and system disk
    storage: local-lvm
    options:
      backup: false
  - size: 400G # VM data disk. Allow default thin provisioning
    storage: local-lvm
    options:
      backup: false
  networks:
  - bridge: vmbr0 # management interface defined by lala100
  - bridge: vmbr1 # replication interface defined by lala100

- vmid: 159 # define the secondary PBS instance on lala150
  name: pbsback
  target_node: lala150 # Host PVE machine name for VM
  memory: 2048
  cores: 2
  disks:
  - size: 30G # VM boot and system disk
    storage: local-lvm
    options:
      backup: false
  - size: 400G # VM data disk. Allow default thin provisioning
    storage: local-lvm
    options:
      backup: false
  networks:
  - bridge: vmbr0 # management interface defined by lala150
  - bridge: vmbr1 # replication interface defined by lala150
